// Get daily name
export const getDailyEsma = () => {
    const today = new Date();
    const dayOfYear = Math.floor((today - new Date(today.getFullYear(), 0, 0)) / (1000 * 60 * 60 * 24));
    return ESMA_UL_HUSNA[dayOfYear % ESMA_UL_HUSNA.length];
